\section{Inventario de productos reproducibles (CSV generados)}
\label{anexo:b}

\subsection{Archivos en \texttt{datos/intermedio/} (entradas de solo lectura, verificadas por MD5)}
\label{anexo:b_intermedio}

\begin{enumerate}
    \item \texttt{contexto\_1PM.csv} -- Población total por municipio, sexo, grupos de edad, áreas.
    \item \texttt{contexto\_1V.csv} -- Viviendas por condición de ocupación, municipio, áreas, tipos.
    \item \texttt{hogares\_1HM\_2HM.csv} -- Hogares con menores 15 y adultos 60+ por municipio, áreas.
    \item \texttt{persona\_demo\_17PM.csv} -- Población 5+ por asistencia escolar/nivel educativo, sexo, municipio, áreas.
    \item \texttt{persona\_social\_5PM.csv} -- Población 10+ por actividad realizada, sexo, municipio, áreas.
    \item \texttt{viviendas\_6V.csv} -- Viviendas ocupadas por material paredes exteriores, municipio, áreas.
    \item \texttt{viviendas\_8V.csv} -- Viviendas ocupadas por disponibilidad servicios, municipio, áreas.
    \item \texttt{viviendas\_10V.csv} -- Personas en hogares por estrato energía, municipio, áreas.
    \item \texttt{ipm\_municipal\_ancho.csv} -- IPM municipal: 15 privaciones + IPM Municipal + IPM (duplicado) + Cabeceras + Centros poblados.
    \item \texttt{universo\_municipal\_maestro.csv} -- Lista maestra 1.122 códigos municipales con nombres, departamentos.
    \item \texttt{control\_nacional.csv} -- Totales nacionales de referencia (población, hogares, viviendas).
\end{enumerate}

    \item \texttt{parametros\_estandarizacion.csv} -- Media y DE por indicador para estandarización (pre-PCA).
\end{enumerate}

\subsection{Archivos en \texttt{resultados/base\_trabajo/} (25 CSV: calidad, sensibilidad, diagnósticos)}
\label{anexo:b_base_trabajo}

\begin{enumerate}
    \item \texttt{especificacion\_indicadores.csv} -- Especificación completa 14 indicadores (fuente, numerador, denominador, dirección, uso).
    \item \texttt{crosswalk\_enlace.csv} -- Equivalencias territoriales para enlace SECOP futuro (94663, 27493, etc.).
    \item \texttt{validaciones.csv} -- (Copia) Validaciones de indicadores y matriz.
    \item \texttt{auditoria\_conteos.csv} -- Auditoría por variable: disponibles, faltantes, negativos, no enteros.
    \item \texttt{areas\_inexistentes.csv} -- Municipios donde no existe Cabecera/Centro poblado/Rural disperso.
    \item \texttt{ausencias\_estructurales\_1PM.csv} -- Combinaciones municipio-área-edad ausentes en 1PM (335, 119 municipios).
    \item \texttt{municipios\_sin\_cabecera.csv} -- Detalle 20 municipios sin cabecera (código, nombre, tipo, población).
    \item \texttt{resumen\_sin\_cabecera\_departamento.csv} -- Agrupación por departamento de municipios sin cabecera.
    \item \texttt{resumen\_hallazgos.csv} -- Resumen cifras de control y hallazgos con estado.
    \item \texttt{covariables\_descriptivas.csv} -- Datos municipales de contexto (población, \% rural, banderas calidad).
    \item \texttt{estadisticos\_indicadores.csv} -- Descriptivos por indicador (rango, media, mediana, asimetría, \% ceros).
    \item \texttt{varianza\_indicadores.csv} -- Varianza por indicador; señalización varianza casi nula.
    \item \texttt{outliers\_z\_mayor\_4.csv} -- Conteo municipios con al menos un $|z|>4$ (32 municipios).
    \item \texttt{parametros\_estandarizacion.csv} -- Media y DE para estandarizar indicadores propuestos.
    \item \texttt{correlaciones\_pares.csv} -- Correlación Pearson para cada par de 7 indicadores (21 pares).
    \item \texttt{pares\_correlacion\_ge\_090.csv} -- Pares con $|r|\ge0,90$ (vacío).
    \item \texttt{pares\_correlacion\_ge\_095.csv} -- Pares con $|r|\ge0,95$ (vacío).
    \item \texttt{diagnosticos\_aptitud.csv} -- KMO global, Bartlett, determinante (diagnósticos previos, no PCA).
    \item \texttt{kmo\_por\_indicador.csv} -- MSA (KMO individual) por indicador (estrato 1--2 $\approx 0,414$).
    \item \texttt{nr\_item\_municipal.csv} -- Valores municipales de ítems NR (con bases y proporciones; estrato ``no tiene'' aparte).
    \item \texttt{sensibilidad\_nr\_municipio.csv} -- Comparación municipal base total vs. base válida (diferencias, marcas).
    \item \texttt{sensibilidad\_nr.csv} -- Resumen sensibilidad NR por indicador.
    \item \texttt{md5\_insumos\_antes.csv} -- Hashes MD5 de 11 archivos entrada antes de ejecución.
    \item \texttt{md5\_insumos\_despues.csv} -- Hashes MD5 después; confirman sin cambios.
\end{enumerate}
