\section{Flujo de procesamiento}
\label{sec:flujo_procesamiento}

\begin{enumerate}
    \item \textbf{Configuración:} \texttt{R/00\_config.R} define todas las rutas, umbrales y banderas.
    \item \textbf{Verificación consolidados:} \texttt{R/01\_verificar\_consolidados.R}
    \begin{enumerate}
        \item Lee los 4 libros Excel CNPV hoja a hoja (como texto, sin chunking real).
        \item Reconstruye \texttt{codigo\_municipio} (5 chars) y valida contra universo maestro.
        \item Genera tablas intermedias \texttt{datos/intermedio/} (contexto, hogares, personas, viviendas, IPM).
        \item Produce inventario, mapa de llaves, cobertura, discrepancias, controles.
        \item Documenta en \texttt{VERIFICACION\_CONSOLIDADOS.md}.
    \end{enumerate}
    \item \textbf{Construcción indicadores:} \texttt{R/02\_base\_final\_indicadores.R}
    \begin{enumerate}
        \item Lee tablas intermedias (solo lectura, verifica MD5).
        \item Calcula denominadores desde 1PM y 1V\_MPIO.
        \item Construye 14 indicadores (7 PCA provisionales + 7 descripción/solapamiento IPM).
        \item Genera versiones base total y base válida para indicadores con NR.
        \item Produce matrices candidatas, \texttt{especificacion\_indicadores.csv}, controles, sensibilidad, \textit{flags}, hashes.
    \end{enumerate}
    \item \textbf{Auditoría estadística:} \texttt{R/03\_auditoria\_indicadores.R}
    \begin{enumerate}
        \item Lee matrices candidatas y calcula descriptivos, correlaciones, KMO, Bartlett, outliers, estandarización.
        \item \textbf{No ejecuta PCA ni bootstrap} (\texttt{ejecutar\_pca = FALSE}).
    \end{enumerate}
    \item \textbf{Prueba de integridad:} \texttt{R/tests/test\_pipeline\_indicadores.R} regenera todo y compara MD5.
\end{enumerate}

\section{Control de integridad}
\label{sec:control_integridad}

\begin{itemize}
    \item \textbf{Hashes MD5:} 11 archivos de entrada monitoreados; hashes antes y después de cada ejecución confirman \textbf{sin modificaciones}.
    \item \textbf{Tolerancias numéricas:} \texttt{tolerancia\_suma = 1e-8}, \texttt{tolerancia\_ipm = 1e-10}, \texttt{tolerancia\_auditoria = 1e-10} (en \texttt{R/00\_config.R}).
    \item \textbf{Controles de consistencia:} Suma grupos edad = total; Hombre + Mujer = Total; Sumas municipales = Control nacional; 17PM total = suma categorías; 5PM total = suma categorías; 8V servicios = suma categorías; 10V BU = suma BV:CD.
    \item \textbf{Refactorización verificada:} Octubre 2026 regeneró 34 CSV idénticos byte-a-byte a la línea base (\texttt{docs/calidad/Refactorizacion\_Pipeline\_Indicadores}).
\end{itemize}

\section{Reproducibilidad}
\label{sec:reproducibilidad_practica}

Para reproducir el estado actual del pipeline desde la raíz del repositorio:

\begin{verbatim}
# 1. Verificación consolidados (genera datos/intermedio/, resultados/verificacion_consolidados/)
Rscript R/01_verificar_consolidados.R

# 2. Construcción indicadores y diagnósticos base (genera datos/trabajo/, resultados/base_trabajo/)
Rscript R/02_base_final_indicadores.R

# 3. Auditoría estadística preliminar (genera resultados/auditoria_estadistica/)
Rscript R/03_auditoria_indicadores.R

# 4. Prueba de integridad (opcional, verifica regeneración byte-a-byte)
Rscript R/tests/test_pipeline_indicadores.R
\end{verbatim}

\begin{itemize}
    \item Requiere R $\ge$ 4.0 y paquetes: \texttt{readxl, readr, dplyr, tidyr, stringr, purrr, janitor, tibble, lubridate, data.table}.
    \item No hay \texttt{renv.lock} ni \texttt{DESCRIPTION} versionado; versiones de paquetes no fijadas (deuda técnica documentada).
    \item Los archivos Excel fuente en \texttt{datos/} \textbf{no están versionados} (ver \texttt{.gitignore}); deben obtenerse del DANE.
    \item La bóveda Obsidian (\texttt{.obsidian/}) contiene notas metodológicas enlazadas (\texttt{Inicio\_Proyecto.md}, \texttt{Final\_Base.md}, \texttt{VERIFICACION\_CONSOLIDADOS.md}, etc.) que documentan decisiones y vinculan a archivos reproducibles.
\end{itemize}
